{# sparkline(points, width=90, height=24) — compact SVG line with soft area. #} {%- macro sparkline(points, width=90, height=24) -%} {%- if points and points|length > 1 -%} {%- set xs = points -%} {%- set ymin = xs|min -%} {%- set ymax = xs|max -%} {%- set span = (ymax - ymin) if (ymax - ymin) > 0 else 1 -%} {%- set n = xs|length -%} {%- set stepx = width / (n - 1) -%} {%- set path -%} {%- for v in xs -%} {%- set x = loop.index0 * stepx -%} {%- set y = height - ((v - ymin) / span) * (height - 2) - 1 -%} {{ "M" if loop.first else "L" }}{{ "%.2f"|format(x) }} {{ "%.2f"|format(y) }}{% if not loop.last %} {% endif %} {%- endfor -%} {%- endset -%} {%- set area = path + ' L' + "%.2f"|format(width) + ' ' + "%.2f"|format(height) + ' L0 ' + "%.2f"|format(height) + ' Z' -%} {%- else -%} {%- endif -%} {%- endmacro -%}
